cmake_minimum_required(VERSION 3.16)
project(LogFlowV3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(logflow src/main.cpp)
target_include_directories(logflow PRIVATE src)
add_executable(logflow_tests tests/tests.cpp)
target_include_directories(logflow_tests PRIVATE src)
enable_testing()
add_test(NAME LogFlowUnitAndIntegrationTests COMMAND logflow_tests)
